\begin{frame}{集合间的关系}

集合间的基本关系：
\begin{description}
	\item<1->[子集：] $\forall x \in A,x \in B$，则称$A$是$B$的子集。
	\item<2->[真子集：] $A\subseteq B, \exists x \in B ,x \notin A$,则$A$是$B$的真子集。
	\item<3->[集合相等：] $A \subseteq B, B \subseteq A,$则 $A = B$。
	\item<4->[空集：] $\phi $是任意集合的子集，是任意非空集合的真子集。
\end{description}

\begin{frame}{集合的基本运算}
	
	集合的运算：
	\begin{enumerate}
		\item 并集： $A \cup B = \{ x | x \in A,\text{或} x \in B\}$.
		\item 交集： $A \cap B = \{ x | x \in A ,\text{且} x \in B \}$.
		\item 补集： $\complement_UA = \{ x | x \in U, \text{且} x \notin A\}$.
	\end{enumerate}
